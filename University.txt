import json

class Student:
    def __init__(self, name, age, group):
        self.name = name
        self.age = age
        self.group = group
    def to_dict(self):
        return {"name": self.name, "age": self.age, "group": self.group}

class Teacher:
    def __init__(self, name, subject):
        self.name = name
        self.subject = subject
    def to_dict(self):
        return {"name": self.name, "subject": self.subject}

class University:
    def __init__(self):
        self.students = []
        self.teachers = []
    def add_student(self, student):
        self.students.append(student)
    def add_teacher(self, teacher):
        self.teachers.append(teacher)
    def save(self, filename):
        data = {
            "students": [s.to_dict() for s in self.students],
            "teachers": [t.to_dict() for t in self.teachers],
        }
        with open(filename, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)
    def load(self, filename):
        with open(filename, "r", encoding="utf-8") as f:
            data = json.load(f)
        self.students = [Student(**s) for s in data.get("students", [])]
        self.teachers = [Teacher(**t) for t in data.get("teachers", [])]
    def show(self):
        if not self.students and not self.teachers:
            print("Пусто")
            return
        for s in self.students:
            print(f"Студент: {s.name}, {s.age}, {s.group}")
        for t in self.teachers:
            print(f"Преподаватель: {t.name}, {t.subject}")

if __name__ == "__main__":
    uni = University()
    filename = "university.json"
    while True:
        cmd = input("R-читать, W-записать, A-добавить, S-показать, Q-выход: ").strip().lower()
        if cmd == "r":
            try:
                uni.load(filename)
                print("Данные загружены")
            except FileNotFoundError:
                print("Файл не найден")
            except json.JSONDecodeError as e:
                print("Ошибка в JSON:", e)
        elif cmd == "w":
            uni.save(filename)
            print("Данные сохранены")
        elif cmd == "a":
            kind = input("Кого добавить? (S - студент, T - преподаватель): ").strip().lower()
            if kind == "s":
                name = input("Имя: ")
                age = int(input("Возраст: "))
                group = input("Группа: ")
                uni.add_student(Student(name, age, group))
                print("Студент добавлен")
            elif kind == "t":
                name = input("Имя: ")
                subject = input("Предмет: ")
                uni.add_teacher(Teacher(name, subject))
                print("Преподаватель добавлен")
            else:
                print("Неизвестный тип")
        elif cmd == "s":
            uni.show()
        elif cmd == "q":
            break
        else:
            print("Неизвестная команда")
